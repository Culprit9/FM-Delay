cmake_minimum_required(VERSION 3.22)
project(FMDelay VERSION 0.11.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "Static Windows runtime")
 add_compile_options(/utf-8)
endif()
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET 11.0 CACHE STRING "Minimum macOS version" FORCE)
endif()
set(FM_FORMATS VST3 Standalone)
if(APPLE)
 list(APPEND FM_FORMATS AU)
endif()
if(DEFINED JUCE_PATH)
 add_subdirectory("${JUCE_PATH}" juce)
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(FMDelay COMPANY_NAME "FM Audio" BUNDLE_ID "audio.fm.delay" PLUGIN_MANUFACTURER_CODE FmAu PLUGIN_CODE FmDl FORMATS ${FM_FORMATS} PRODUCT_NAME "FM DELAY" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Delay)
juce_add_binary_data(FMArtwork SOURCES Assets/panel.png)
juce_generate_juce_header(FMDelay)
target_sources(FMDelay PRIVATE Source/Plugin.cpp)
target_compile_definitions(FMDelay PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(FMDelay PRIVATE FMArtwork juce::juce_audio_utils juce::juce_recommended_config_flags)
add_executable(dsp_tests Tests/DSPTests.cpp)
enable_testing()
add_test(NAME DSP COMMAND dsp_tests)
add_executable(preset_tests Tests/PresetTests.cpp)
add_test(NAME Presets COMMAND preset_tests)
add_executable(regression_tests Tests/RegressionTests.cpp)
add_test(NAME Regressions COMMAND regression_tests)
add_executable(voicing_tests Tests/VoicingTests.cpp)
add_test(NAME Voicing COMMAND voicing_tests)
add_executable(wet_diffusion_tests Tests/WetDiffusionTests.cpp)
add_test(NAME WetDiffusion COMMAND wet_diffusion_tests)
add_executable(character_duck_tests Tests/CharacterDuckTests.cpp)
add_test(NAME CharacterDucking COMMAND character_duck_tests)
add_executable(quality_benchmark Tests/QualityBenchmark.cpp)
add_executable(ping_pong_tests Tests/PingPongTests.cpp)
add_test(NAME PingPong COMMAND ping_pong_tests)
add_executable(texture_tests Tests/TextureTests.cpp)
add_test(NAME Texture COMMAND texture_tests)

option(FM_BUILD_HOST_CHECK "Build isolated VST3 host validation tool" OFF)
if(FM_BUILD_HOST_CHECK)
 juce_add_console_app(FMHostCheck)
 juce_generate_juce_header(FMHostCheck)
 target_sources(FMHostCheck PRIVATE Tests/HostCheck.cpp)
 target_compile_definitions(FMHostCheck PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(FMHostCheck PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
 if(APPLE)
  add_executable(FMDirectAUCheck Tests/DirectAUCheck.cpp)
  target_link_libraries(FMDirectAUCheck PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
 endif()
endif()
add_executable(tone_filter_tests Tests/ToneFilterTests.cpp)
add_test(NAME ToneFilters COMMAND tone_filter_tests)
